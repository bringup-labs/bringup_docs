# Bringup Docs Refactor Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Replace the Bagmaster-era documentation with docs for the Bringup desktop app — task-centric tutorials, per-extension reference pages, generated screenshots, and a FAQ.

**Architecture:** Docusaurus 3.9 monorepo (pnpm workspace: root + `website/`). Content lives in `docs/` and is served at `routeBasePath: '/'` with an autogenerated sidebar, so directory layout *is* navigation and `_category_.json` files control ordering. Screenshots are produced by a Playwright/Electron spec in the sibling `bringup_desktop_app` repo that writes PNGs directly into this repo.

**Tech Stack:** Docusaurus 3.9.2, React 19, TypeScript, pnpm 10.5.2, Playwright + Electron (screenshots).

**Spec:** [`specs/2026-07-25-bringup-docs-refactor-design.md`](../specs/2026-07-25-bringup-docs-refactor-design.md)

---

## A note on step granularity

Code steps (config, components, the Playwright spec) contain complete, paste-ready code.

Prose pages do **not** have their finished markdown inlined — 25 pages of final copy inside a plan is writing the docs twice, and it rots the moment the first page is edited. Each prose task instead specifies: exact path, exact frontmatter, the section outline, **the grounding source for every factual claim** (`file:line`), and acceptance checks. Do not invent facts not traceable to a listed source; where a source does not exist, the plan says to write a marked `TODO` instead.

```ts
/**
 * Generates the screenshots used by the docs site.
 *
 * NOT a test — it asserts nothing about behaviour. It drives the app into each
 * documented state and writes a PNG. It lives in the e2e suite because that is
 * where the Electron fixtures are; it is excluded from `npm run test:e2e` by
 * filename (specs are selected explicitly there).
 *
 * Run: npx playwright test e2e/screenshots.spec.ts
 * Output: ../bringup_docs/docs/assets/screenshots/
 */
import { mkdirSync } from 'fs'
import { join } from 'path'
import { test } from './fixtures'
import { buildSignedInSecrets } from './seed-session'

const OUT_DIR = join(__dirname, '..', '..', 'bringup_docs', 'docs', 'assets', 'screenshots')

const ORG_ID = 'org-docs-demo'
const ORG_NAME = 'Demo Robotics'

mkdirSync(OUT_DIR, { recursive: true })

const shot = (name: string) => join(OUT_DIR, `${name}.png`)

test.describe('first-run screens', () => {
    test.use({ onboardingCompleted: false })

    test('welcome and daemon setup', async ({ window }) => {
        await window.waitForLoadState('networkidle')
        await window.screenshot({ path: shot('welcome-screen') })

        // WelcomeStep -> SetupStep is a single forward button.
        await window.getByRole('button', { name: /get started|next|continue/i }).first().click()
        await window.waitForTimeout(600) // framer-motion transition is 250ms
        await window.screenshot({ path: shot('daemon-setup') })
    })
})

/** Signed-in session on disk, mirroring e2e/offline.spec.ts. */
const signedIn = test.extend<{}, { seededSecrets: Buffer | null }>({
    seededSecrets: [
        async ({}, use) => {
            await use(
                await buildSignedInSecrets({
                    name: 'Docs Demo',
                    email: 'docs@bringup.dev',
                    expiresAt: Date.now() + 3_600_000, // future: warm start, no refresh call
                }),
            )
        },
        { scope: 'worker' },
    ],
    seedFiles: async ({ seededSecrets }, use) => {
        const files: Record<string, string | Buffer> = {
            'orgs.json': JSON.stringify({
                activeProfileId: ORG_ID,
                profiles: [
                    {
                        id: ORG_ID,
                        name: ORG_NAME,
                        managementUrl: 'https://api.bringup.dev',
                        enrolled: true,
                    },
                ],
            }),
        }
        if (seededSecrets) files['secrets.enc'] = seededSecrets
        await use(files)
    },
})

signedIn.describe('workbench screens', () => {
    signedIn.use({ onboardingCompleted: true })

    signedIn('workbench, palette, extensions, settings', async ({ seededSecrets, window }) => {
        signedIn.skip(seededSecrets === null, 'platform has no usable keyring; cannot seed a session')

        await window.waitForLoadState('networkidle')
        await window.screenshot({ path: shot('workbench') })

        // Command palette.
        await window.keyboard.press(process.platform === 'darwin' ? 'Meta+Shift+P' : 'Control+Shift+P')
        await window.waitForTimeout(400)
        await window.screenshot({ path: shot('command-palette') })
        await window.keyboard.press('Escape')

        // Rail (icon-only) sidebar mode, persisted under `bu-sidebar-mode`.
        await window.evaluate(() => localStorage.setItem('bu-sidebar-mode', 'rail'))
        await window.reload()
        await window.waitForLoadState('networkidle')
        await window.screenshot({ path: shot('workbench-rail-sidebar') })
        await window.evaluate(() => localStorage.setItem('bu-sidebar-mode', 'labeled'))
        await window.reload()
        await window.waitForLoadState('networkidle')
    })
})
```
